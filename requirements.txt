# Jarvis Agentic OS - runtime dependencies
#
# Everything here is optional in the sense that jarvis.py boots and runs the
# full core without it; each package only enables one optional capability and
# is imported lazily. Nothing in this list downloads a model on first boot.

# --- Core (already present in most environments) -------------------------
psutil>=5.9            # process supervision for the software factory / daemon

# --- Language layer (optional) -------------------------------------------
langdetect>=1.0.9      # script/language detection for the language layer
argostranslate>=1.9    # OFFLINE translation. Requires language pairs to be
                       # installed separately and explicitly:
                       #     from jarvis_os.language import install_argos_packages
                       #     install_argos_packages(approved=True)
                       # Jarvis never calls this on its own.

# --- Voice interface (optional) ------------------------------------------
speech_recognition>=3.10   # offline speech-to-text (Sphinx/Whisper backend)
pyttsx3>=2.90              # offline text-to-speech output

# --- Device sync (optional) ----------------------------------------------
# Powers `python jarvis.py --device serve`: a local-only WSS listener with a
# per-install certificate and pairing token (both generated into data/ on
# first start, never downloaded). Imported lazily, so the core boots without
# them; `--device` degrades to an actionable error until they are installed.
websockets>=12.0           # WSS transport for the one-directional sync server
cryptography>=42.0         # local TLS certificate + key generation (data/tls)

# --- Background voice daemon (optional) -----------------------------------
# Enables `python jarvis.py --voice` and setup-windows-daemon.ps1:
# a non-blocking mic stream feeding an offline Vosk wake-word recogniser.
# The Vosk model itself is NOT a pip package - the setup script downloads
# it into models/ (Jarvis never downloads models at runtime).
vosk>=0.3.45
sounddevice>=0.4.6

# --- OS UI control (optional) ---------------------------------------------
# Powers the gated `ui_control` tool (click/type/hotkey/scroll/screenshot and
# the ocr / click_text actions behind `--control gui`).
# Imported lazily inside jarvis_os/ui.py, so the core boots without it.
pyautogui>=0.9.54
pytesseract>=0.3.10         # Python wrapper only: the Tesseract OCR ENGINE must
                            # be installed separately for `ocr` to work; ui_control
                            # degrades to an actionable error until then.

# --- Local translation fallback (optional, heavy) ------------------------
# Only needed if you want the neural offline translator instead of Argos.
# It is what serves Tamil: NLLB-200 covers `ta` natively, while Argos has no
# en->ta package. ~2.4 GB of weights land in language.cache_dir (data/langcache)
# and are never downloaded automatically -- Jarvis loads with
# local_files_only=True, so fetch them explicitly first.
transformers>=4.40
sentencepiece>=0.1.99    # NLLB tokenizer
torch>=2.0

# --- Terminal dashboard (optional) ----------------------------------------
# Powers `python jarvis.py --dashboard` and the `dashboard` REPL command: a
# Rich Panel over the plain-text renderer. Imported lazily, so the core and
# the dashboard's plain-text fallback work without it.
rich>=13

# NOTE: PyAudio is required by SpeechRecognition for microphone capture on most
# systems, but it needs PortAudio binaries, so it is not listed here. Install it
# only if you enable voice:
#     pip install pyaudio
# (sounddevice, used by the background daemon, ships its own PortAudio.)
# Voice degrades gracefully to text when any of these are absent.
